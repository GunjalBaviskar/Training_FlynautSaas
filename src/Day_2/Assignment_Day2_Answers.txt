// Create a program that asks the user to input a year andchecks if it is a leap year: 
	A year is a leap year if it is divisible by 4 but not divisible by 100,except if it is divisible by 400. 

import java.util.Scanner;

public class Main {
  public static void main(String[] args) { 
    Scanner scanner = new Scanner(System.in);  
	System.out.print("Enter a year: ");
    int year = scanner.nextInt();
    
   	if( year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)) {
    	System.out.println(year + " is a leap year");
    } else {
     	System.out.println(year + " is not aleap year");
    }
    scanner.close();
  }
}
